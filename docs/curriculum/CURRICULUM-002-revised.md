```typescript
interface Lesson {
  id: string;
  kapitelId: string;
  levelId: string;
  title: string;
  order: number;
  duration: number;     // 10, 12, 15, or 20 minutes
  
  // Grammar focus
  grammarFocus: string[];
  
  // Learning content

  grammar: GrammarRule[];
  listening: ListeningExercise[];

  writing: WritingExercise[];
  speaking: SpeakingExercise[];
  
  // Exercises
  exercises: Exercise[];

}
```
